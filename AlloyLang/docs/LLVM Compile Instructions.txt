git clone --config core.autocrlf=false https://github.com/llvm/llvm-project.git

# From Visual Studio x64 command line
cd \projects\llvm-project
cmake -S llvm -B build -G "Visual Studio 17 2022" -DLLVM_ENABLE_PROJECTS="lld;clang" -DCMAKE_INSTALL_PREFIX="c:\llvm" -DCMAKE_BUILD_TYPE=debug -DLLVM_TARGETS_TO_BUILD=X86 -Thost=x64
cmake --build build --parallel 4
cmake --install build --config Debug

C:\Projects\llvm-project\build\Debug\bin\clang hello.c -o hello.exe -DLLVM_USE_LINKER=lld

# Generate LLVM bitcode file
clang -O3 -emit-llvm hello.c -c -o hello.bc
# run it with:
lli hello.bc
