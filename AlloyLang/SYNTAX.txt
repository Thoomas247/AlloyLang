(* -- TOKEN DEFINITIONS -- *)

	identifier:						( letter | '_' ) { letter | digit | '_' } ;

	comment:						('//' { character } '\n') | ('/*' { character } '*/') ;

	export_label:					'exp' ;
	public_label:					'pub' ;

	extern_keyword:					'extern' ;
	struct_keyword:					'struct' ;
	enum_keyword:					'enum' ;
	function_keyword:				'fn' ;

	type_keyword:					'type' ;

	variable_keyword:				'var' ;
	constant_keyword:				'const' ;

	resource_keyword:				'resource' ;
	component_keyword:				'component' ;
	query_keyword:					'query' ;
	system_keyword:					'system' ;
	group_keyword:					'group' ;
	application_keyword:			'application' ;

	require_keyword:				'require' ;
	exclude_keyword:				'exclude' ;

	for_keyword:					'for' ;
	while_keyword:					'while' ;
	if_keyword:						'if' ;
	else_keyword:					'else' ;
	return_keyword:					'return' ;

	new_keyword:					'new' ;
	move_keyword:					'move' ;

	reference:						'&' ;
	pointer:						'*' ;

	comma:							',' ;
	colon:							':' ;
	semicolon:						';' ;
	dot:							'.' ;
	arrow:							'->' ;
	ellipsis:						'...' ;

	open_paren:						'(' ;
	close_paren:					')' ;
	open_brace:						'{' ;
	close_brace:					'}' ;
	open_bracket:					'[' ;
	close_bracket:					']' ;

	assignment_operator:			'=' ;

	logical_operator:				'&&' | '||' ;
	relational_operator:			'==' | '!=' | '>' | '<' | '>=' | '<=' ;
	additive_operator:				'+' | '-' ;
	multiplicative_operator:		'*' | '/' | '%' ;

	unary_operator:					'!' | '-' ;

	integer_literal:				(* digit regex *) { (* digit regex *) } ;
	float_literal:					(* digit regex *) { (* digit regex *) } '.' (* digit regex *) { (* digit regex *) } ;
	boolean_literal:				'true' | 'false' ;
	string_literal:					'"' { (* character regex *) } '"' ;
	character_literal:				'\'' [ (* character regex *) ] '\'' ;

(* -- AST NODE DEFINITIONS -- *)


	(* - Types - *)

	TYPE:						[ reference | pointer ] ( NAMED_TYPE | STRUCT_TYPE | ARRAY_TYPE ) ;

		NAMED_TYPE:					identifier ;
		STRUCT_TYPE:				struct_keyword open_brace [ STRUCT_MEMBER { comma STRUCT_MEMBER } ] close_brace ;
			STRUCT_MEMBER:				identifier colon TYPE ;
		ARRAY_TYPE:					open_bracket TYPE [ semicolon integer_literal ] close_bracket ;

	NAMED_TYPE_DEFINITION:		type_keyword identifier assignment_operator TYPE semicolon ;


	(* - Variables - *)

	NAMED_VARIABLE:				identifier ;
	NAMED_VARIABLE_DECLARATION	( variable | constant ) identifier colon TYPE ;
	NAMED_VARIABLE_DEFINITION	NAMED_VARIABLE_DECLARATION assignment_operator EXPRESSION ;


	(* - Functions - *)

	NAMED_FUNCTION:				identifier ;
	NAMED_FUNCTION_DEFINITION:	FUNCTION_SIGNATURE STATEMENT_BLOCK ;
	EXTERN_FUNCTION_DEFINITION:	extern_keyword FUNCTION_SIGNATURE semicolon ;

	FUNCTION_SIGNATURE:			function identifier open_paren [ NAMED_VARIABLE_DECLARATION { comma NAMED_VARIABLE_DECLARATION } ] close_paren [ arrow RETURN_TYPE ] ;
		RETURN_TYPE:				[ variable | constant ] TYPE ;

	FUNCTION_CALL:				NAMED_FUNCTION open_paren [ EXPRESSION { comma EXPRESSION } ] close_paren ;

	(* - Expressions - *)

	EXPRESSION:			PRIMARY 
						| POSTFIX 
						| UNARY 
						| BINARY 
						| ASSIGNMENT
						;

		PRIMARY:			LITERAL 
							| NAMED_VARIABLE
							| NAMED_VARIABLE_DEFINITION
							| FUNCTION_CALL 
							| CONSTRUCTOR
							| POINTER_INIT
							| POINTER_MOVE
							| INITIALIZER_LIST
							| ENCLOSED_EXPRESSION
							; 
			LITERAL:				integer_literal | float_literal | boolean_literal | string_literal | character_literal ;
			CONSTRUCTOR:			NAMED_TYPE open_brace [ identifier assignment_operator EXPRESSION { comma identifier assignment_operator EXPRESSION } ] close_brace ;
			POINTER_INIT:			new_keyword ( EXPRESSION | ( open_bracket EXPRESSION semicolon EXPRESSION close_bracket ) ) ;
			POINTER_MOVE:			move_keyword NAMED_VARIABLE ;
			INITIALIZER_LIST:		open_brace [ EXPRESSION { comma EXPRESSION } ] close_brace ;
			ENCLOSED_EXPRESSION:	open_paren EXPRESSION close_paren ;

		POSTFIX:		ARRAY_ACCESS | MEMBER_ACCESS ;
			ARRAY_ACCESS:			EXPRESSION open_bracket EXPRESSION close_bracket ;
			MEMBER_ACCESS:			EXPRESSION dot identifier ;
			

		UNARY:			unary_operator EXPRESSION ;
		BINARY:			EXPRESSION binary_operator EXPRESSION ;

		ASSIGNMENT:		ASSIGNABLE assignment_operator EXPRESSION ;
			ASSIGNABLE:				NAMED_VARIABLE | POSTFIX ;


	(* - Statements - *)

	STATEMENT:		( NAMED_VARIABLE_DEFINITION semicolon )
					| ( FUNCTION_CALL semicolon )
					| ( ASSIGNMENT semicolon )
					| FOR_LOOP
					| WHILE_LOOP
					| IF_STATEMENT
					| STATEMENT_BLOCK
					| RETURN
					;

		FOR_LOOP:			for_keyword open_paren EXPRESSION semicolon EXPRESSION semicolon EXPRESSION close_paren STATEMENT ;
		WHILE_LOOP:			while_keyword ENCLOSED_EXPRESSION STATEMENT ;
		IF_STATEMENT:		if_keyword ENCLOSED_EXPRESSION STATEMENT [ else_keyword STATEMENT ] ;
		STATEMENT_BLOCK:	open_brace { STATEMENT } close_brace ;
		RETURN:				return_keyword [ EXPRESSION ] semicolon ;


	(* - ECS Constructs - *)

	NAMED_COMPONENT:			identifier ;
	NAMED_COMPONENT_DEFINITION:	component_keyword identifier assignment_operator TYPE semicolon ;

	NAMED_RESOURCE:				identifier ;
	NAMED_RESOURCE_DEFINITION:	resource_keyword identifier assignment_operator TYPE semicolon ;

	NAMED_QUERY:				identifier ;
	NAMED_QUERY_DEFINITION:		query_keyword identifier open_brace { ( var_keyword | const_keyword ) QUERYABLE semicolon } close_brace ;
		QUERYABLE:					NAMED_COMPONENT | NAMED_RESOURCE ;

	NAMED_SYSTEM:				identifier ;
	NAMED_SYSTEM_DEFINITION:	system_keyword identifier open_paren [ NAMED_QUERY { comma NAMED_QUERY } ] close_paren STATEMENT_BLOCK ;

	NAMED_GROUP:				identifier ;
	NAMED_GROUP_DEFINITION:		group_keyword identifier open_brace NAMED_SYSTEM { comma NAMED_SYSTEM } close_brace ;

	APPLICATION_DEFINITION:		application_keyword identifier open_brace { ( 'start' | 'update' | 'end' ) NAMED_GROUP_LIST } close_brace ;
		NAMED_GROUP_LIST:			open_brace [ NAMED_GROUP { comma NAMED_GROUP } ] close_brace ;


	(* - Definitions - *)

	DEFINITION:		NAMED_TYPE_DEFINITION
					| NAMED_VARIABLE_DEFINITION
					| NAMED_FUNCTION_DEFINITION
					| EXTERN_FUNCTION_DEFINITION
					| NAMED_COMPONENT_DEFINITION
					| NAMED_RESOURCE_DEFINITION
					| NAMED_QUERY_DEFINITION
					| NAMED_SYSTEM_DEFINITION
					| NAMED_GROUP_DEFINITION
					| APPLICATION_DEFINITION
					;
