(* -- TOKEN DEFINITIONS -- *)

identifier:					( letter | '_' ) { letter | digit | '_' } ;

export_label:				'exp' ;
public_label:				'pub' ;

struct_keyword:				'struct' ;
enum_keyword:				'enum' ;
function_keyword:			'fn' ;

variable_keyword:			'var' ;
constant_keyword:			'const' ;

for_keyword:				'for' ;
while_keyword:				'while' ;
if_keyword:					'if' ;
else_keyword:				'else' ;
return_keyword:				'return' ;

reference:					'&' ;
pointer:					'*' ;

comma:						',' ;
colon:						':' ;
semicolon:					';' ;
arrow:						'->' ;

open_paren:					'(' ;
close_paren:				')' ;
open_brace:					'{' ;
close_brace:				'}' ;

assignment_operator:		'=' ;

logical_operator:			'&&' | '||' ;
relational_operator:		'==' | '!=' | '>' | '<' | '>=' | '<=' ;
additive_operator:			'+' | '-' ;
multiplicative_operator:	'*' | '/' | '%' ;

unary_operator:				'!' | '@' ;

integer_literal:			(* digit regex *) { (* digit regex *) } ;
float_literal:				(* digit regex *) { (* digit regex *) } '.' (* digit regex *) { (* digit regex *) } ;
boolean_literal:			'true' | 'false' ;
string_literal:				'"' { (* character regex *) } '"' ;
character_literal:			'\'' [ (* character regex *) ] '\'' ;

(* -- AST NODE DEFINITIONS -- *)

LITERAL:					integer_literal | float_literal | boolean_literal | string_literal | character_literal ;


TYPE_IDENTIFIER:			[ reference | pointer ] identifier ;


TYPE_DECLARATION:			[ variable | constant ] TYPE_IDENTIFIER ;


VALUE_DECLARATION:			[ variable | constant ] identifier colon TYPE_IDENTIFIER ;


VALUE_DEFINITION:			VALUE_DECLARATION assignment_operator EXPRESSION semicolon ;
STRUCT_DEFINITION:			struct identifier open_brace { VALUE_DECLARATION semicolon } close_brace ; (* TODO: force default values? *)
ENUM_DEFINITION:			enum identifier open_brace identifier { comma identifier } close_brace ;
FUNCTION_DEFINITION:		function identifier open_paren [ VALUE_DECLARATION { comma VALUE_DECLARATION } ] close_paren [ arrow TYPE_DECLARATION ] BLOCK_STATEMENT ;


DEFINITION:					VALUE_DECLARATION | STRUCT_DEFINITION | ENUM_DEFINITION | FUNCTION_DEFINITION ;
QUALIFIED_DEFINITION:		[ export | public ] DEFINITION ;


FUNCTION_CALL_EXPRESSION:	identifier open_paren [ EXPRESSION { comma EXPRESSION } ] close_paren ;
ENCLOSED_EXPRESSION:		open_paren EXPRESSION close_paren ;

MULTIPLICATIVE_EXPRESSION:	ADDITIVE_EXPRESSION [ multiplicative_operator MULTIPLICATIVE_EXPRESSION ] ;
ADDITIVE_EXPRESSION:		RELATIONAL_EXPRESSION [ additive_operator ADDITIVE_EXPRESSION ] ;
RELATIONAL_EXPRESSION:		LOGICAL_EXPRESSION [ relational_operator RELATIONAL_EXPRESSION ] ;
LOGICAL_EXPRESSION:			PRIMARY_EXPRESSION [ logical_operator LOGICAL_EXPRESSION ] ;
PRIMARY_EXPRESSION:			identifier | LITERAL | FUNCTION_CALL_EXPRESSION | ENCLOSED_EXPRESSION ;

ASSIGNMENT_EXPRESSION:		identifier assignment_operator EXPRESSION ;

EXPRESSION:					MULTIPLICATIVE_EXPRESSION | ASSIGNMENT_EXPRESSION ;


ASSIGNMENT_STATEMENT:		ASSIGNMENT_EXPRESSION semicolon ;
FOR_LOOP_STATEMENT:			for open_paren EXPRESSION semicolon EXPRESSION semicolon EXPRESSION close_paren STATEMENT ;
WHILE_LOOP_STATEMENT:		while ENCLOSED_EXPRESSION STATEMENT ;
IF_STATEMENT:				if ENCLOSED_EXPRESSION STATEMENT [ else STATEMENT ] ;
BLOCK_STATEMENT:			open_brace {STATEMENT} close_brace ;
RETURN_STATEMENT:			return [ EXPRESSION ] semicolon ;


MODULE:						{ QUALIFIED_DEFINITION } ;


PROGRAM:					{ MODULE } ;
