(* -- TOKEN DEFINITIONS -- *)

	identifier:						( letter | '_' ) { letter | digit | '_' } ;
	long_identifier:				identifier { double_colon identifier } ;

	comment:						('//' { character } '\n') | ('/*' { character } '*/') ;

	extern_keyword:					'extern' ;
	struct_keyword:					'struct' ;
	enum_keyword:					'enum' ;
	function_keyword:				'fn' ;
	macro_keyword:					'macro' ;

	import_keyword:					'import' ;
	as_keyword:						'as' ;

	public_keyword:					'pub' ;
	export_keyword:					'exp' ;

	type_keyword:					'type' ;

	variable_keyword:				'var' ;
	constant_keyword:				'const' ;

	for_keyword:					'for' ;
	while_keyword:					'while' ;
	if_keyword:						'if' ;
	else_keyword:					'else' ;
	switch_keyword:					'switch' ;
	case_keyword:					'case' ;
	return_keyword:					'return' ;

	new_keyword:					'new' ;
	move_keyword:					'move' ;

	pound:							'#' ;
	at_symbol:						'@' ;

	reference:						'&' ;
	pointer:						'*' ;

	comma:							',' ;
	colon:							':' ;
	semicolon:						';' ;
	double_colon:					'::' ;
	dot:							'.' ;
	arrow:							'->' ;
	ellipsis:						'...' ;

	open_paren:						'(' ;
	close_paren:					')' ;
	open_brace:						'{' ;
	close_brace:					'}' ;
	open_bracket:					'[' ;
	close_bracket:					']' ;

	pipe_operator:					'|' ;

	assignment_operator:			'=' ;

	unary_operator:					'!' | '-' | '&' ;
	binary_operator:				'&&' | '||' | '==' | '!=' | '>' | '<' | '>=' | '<=' | '+' | '-' | '*' | '/' | '%' ;

	integer_literal:				(* digit regex *) { (* digit regex *) } ;
	float_literal:					(* digit regex *) { (* digit regex *) } '.' (* digit regex *) { (* digit regex *) } ;
	boolean_literal:				'true' | 'false' ;
	string_literal:					'"' { (* character regex *) } '"' ;
	character_literal:				'\'' [ (* character regex *) ] '\'' ;


	(* - Types - *)

	TYPE:						[ reference | pointer ] ( TYPE_NAME | STRUCT_TYPE | ENUM_TYPE | ARRAY_TYPE | MACRO_CALL ) ;

		TYPE_NAME:					( identifier | long_identifier ) [ open_paren TYPE { comma TYPE } close_paren ] ;
		STRUCT_TYPE:				struct_keyword open_brace [ identifier colon TYPE { comma identifier colon TYPE } ] close_brace ;
		ENUM_TYPE:					enum_keyword open_brace identifier [ colon TYPE ] { comma identifier [ colon TYPE ] } close_brace ;
		ARRAY_TYPE:					open_bracket TYPE [ semicolon LITERAL ] close_bracket ;
		
	TYPE_IDENTIFIER:			identifier [ open_paren GENERIC_PARAMETER { comma GENERIC_PARAMETER } close_paren ] ;
	TYPE_DEFINITION:			type_keyword TYPE_IDENTIFIER assignment_operator TYPE semicolon ;
	

	(* - Variables - *)

	VARIABLE:					identifier

	VARIABLE_DECLARATION:		( variable | constant ) identifier [ colon TYPE ] ;
	VARIABLE_DEFINITION:		VARIABLE_DECLARATION assignment_operator EXPRESSION ;


	(* - Functions - *)

	FUNCTION_PARAMETER:			VARIABLE_DECLARATION | GENERIC_PARAMETER ;

	RETURN_TYPE:				[ variable | constant ] TYPE ;
	FUNCTION_TYPE:				open_paren [ FUNCTION_PARAMETER { comma FUNCTION_PARAMETER } ] close_paren [ arrow RETURN_TYPE ] ;

	FUNCTION_DEFINITION:		function_keyword [ TYPE_IDENTIFIER colon ] identifier FUNCTION_TYPE STATEMENT_BLOCK ;
	EXTERN_DEFINITION:			extern_keyword identifier FUNCTION_TYPE semicolon ;
	
	FUNCTION_CALL:				[ TYPE_NAME colon ] ( identifier | long_identifier ) open_paren [ EXPRESSION { comma EXPRESSION } ] close_paren ; (* TYPE_NAME can contain a variable name in this context *)


	(* Macros *)

	MACRO_VARIABLE_IDENTIFIER:	identifier ;
	MACRO_CALL:					at_symbol ( identifier | long_identifier ) open_paren [ ( MACRO_VARIABLE_IDENTIFIER | MACRO_CALL ) { comma ( MACRO_VARIABLE_IDENTIFIER | MACRO_CALL ) } ] close_paren ;

	MACRO_DEFINITION:			macro_keyword identifier 
								open_paren [ ( type | fn ) identifier { comma ( type | fn ) identifier } ] close_paren 
								[ arrow ( type | fn ) ] 
								open_brace { MACRO_STATEMENT } close_brace ;

	MACRO_RETURN:				return_keyword ( MACRO_VARIABLE_IDENTIFIER | MACRO_CALL | TYPE ) semicolon ;

	MACRO_STATEMENT:			MACRO_DEFINITION | ( MACRO_CALL semicolon ) | TYPE_DEFINITION | FUNCTION_DEFINITION | MACRO_RETURN ;


	(* - Expressions - *)

	EXPRESSION:			PRIMARY 
						| POSTFIX 
						| UNARY 
						| BINARY 
						| ASSIGNMENT
						;

		PRIMARY:			LITERAL 
							| VARIABLE
							| VARIABLE_DEFINITION
							| FUNCTION_CALL 
							| CONSTRUCTOR
							| POINTER_INIT
							| POINTER_MOVE
							| INITIALIZER_LIST
							| ENCLOSED_EXPRESSION
							| ENUM_VALUE
							; 
			LITERAL:				integer_literal | float_literal | boolean_literal | string_literal | character_literal ;
			CONSTRUCTOR:			TYPE_NAME open_brace [ identifier assignment_operator EXPRESSION { comma identifier assignment_operator EXPRESSION } ] close_brace ;
			POINTER_INIT:			new_keyword ( EXPRESSION | ( open_bracket EXPRESSION semicolon EXPRESSION close_bracket ) ) ;
			POINTER_MOVE:			move_keyword VARIABLE ; (* TODO: this should be more than just VARIABLE *)
			INITIALIZER_LIST:		open_brace [ EXPRESSION { comma EXPRESSION } ] close_brace ;
			ENCLOSED_EXPRESSION:	open_paren EXPRESSION close_paren ;
			ENUM_VALUE:				TYPE_NAME pipe_operator identifier [ open_paren EXPRESSION close_paren ] ;

		POSTFIX:		ARRAY_ACCESS | MEMBER_ACCESS ;
			ARRAY_ACCESS:			EXPRESSION open_bracket EXPRESSION close_bracket ;
			MEMBER_ACCESS:			EXPRESSION dot identifier ;
			

		UNARY:			unary_operator EXPRESSION ;
		BINARY:			EXPRESSION binary_operator EXPRESSION ;

		ASSIGNMENT:		EXPRESSION assignment_operator EXPRESSION ;


	(* - Statements - *)

	STATEMENT:		( VARIABLE_DEFINITION semicolon )
					| ( FUNCTION_CALL semicolon )
					| ( ASSIGNMENT semicolon )
					| FOR_LOOP
					| WHILE_LOOP
					| IF_STATEMENT
					| SWITCH_STATEMENT
					| STATEMENT_BLOCK
					| RETURN
					;

		FOR_LOOP:			for_keyword open_paren EXPRESSION semicolon EXPRESSION semicolon EXPRESSION close_paren STATEMENT ;
		WHILE_LOOP:			while_keyword ENCLOSED_EXPRESSION STATEMENT ;
		IF_STATEMENT:		if_keyword ENCLOSED_EXPRESSION STATEMENT [ else_keyword STATEMENT ] ;
		SWITCH_STATEMENT:	switch_keyword ENCLOSED_EXPRESSION open_brace 
							{ case_keyword ENCLOSED_EXPRESSION [ arrow identifier ] STATEMENT } 
							close_brace ;
		STATEMENT_BLOCK:	open_brace { STATEMENT } close_brace ;
		RETURN:				return_keyword [ EXPRESSION ] semicolon ;

	
	(* - Annotations - *)

	ANNOTATION:					pound open_bracket identifier [ open_paren [ identifier { comma identifier } ] ] close_bracket ;


	(* - Modules - *)

	MODULE_IMPORT:				import_keyword ( identifier | long_identifier ) [ as_keyword ( identifier | long_identifier ) ] semicolon ;
